=====================Temperature(DS18B20)===================
#include <ESP8266WiFi.h>
#include <ThingSpeak.h>

#include <DallasTemperature.h>
#include <OneWire.h>

#define ONE_WIRE_BUS 16 // pin for Arduino*********
#define ONE_WIRE_BUS D0 // pin for nodeMCU*********
#define LED_PIN D1      // Pin for LED*************

const char* ssid = "Wokwi-GUEST"; // WiFi SSID
const char* password = "";        // WiFi Password
unsigned long myChannelNo = 2762404; // ThingSpeak Channel Number
const char* myWriteAPIkey = "MY_API_KEY"; // ThingSpeak Write API Key

const float temperatureThreshold = 30.0; // Set threshold in °C

WiFiClient client; 
OneWire oneWire(ONE_WIRE_BUS); 
DallasTemperature sensors(&oneWire);

//LED(short leg) → 220Ω resistor → GND*******************************************
//connect power to data with resistor D0<-->3V3 through resistor*****************
void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  sensors.begin();
  
  // Connecting to WiFi
  Serial.print("Connecting to WiFi");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.print(".");
  }
  Serial.println("\nConnected to WiFi");
  Serial.print("Connected to network: ");
  Serial.println(WiFi.SSID());
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());
  ThingSpeak.begin(client);
}

void loop() {
  Serial.print("Requesting temperature... ");
  sensors.requestTemperatures();
  float temperatureC = sensors.getTempCByIndex(0);
  Serial.print("Temperature: ");
  Serial.println(temperatureC);
  if (temperatureC > temperatureThreshold) {
    digitalWrite(LED_PIN, HIGH); // Turn on LED
    Serial.println("Temperature above threshold! LED ON.");
  } else {
    digitalWrite(LED_PIN, LOW); // Turn off LED
    Serial.println("Temperature below threshold. LED OFF.");
  }

  int statusCode = ThingSpeak.writeField(myChannelNo, 1, temperatureC, myWriteAPIkey);
  if (statusCode == 200) {
    Serial.println("Successfully updated ThingSpeak");
  } else {
    Serial.print("Problem updating channel. HTTP error code: ");
    Serial.println(statusCode);
  }
  delay(2000);
}

=====================UltraSonic===================
#include <ESP8266WiFi.h>
#include <ThingSpeak.h>

#define TRIG_PIN D6  // Trigger Pin
#define ECHO_PIN D5  // Echo Pin
#define LED_PIN D4   // LED Pin
#define BUZZER_PIN D3 // Buzzer Pin

const char* ssid = "Wokwi-GUEST";  // WiFi SSID
const char* password = "";         // WiFi Password
unsigned long myChannelNo = 2762405; // ThingSpeak Channel Number
const char* myWriteAPIkey = "YOUR_API_KEY"; // Replace with your ThingSpeak Write API Key

WiFiClient client;

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
  digitalWrite(BUZZER_PIN, LOW);

  Serial.print("Connecting to WiFi");
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected");
  Serial.print("Connected to network: ");
  Serial.println(WiFi.SSID());
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());
  ThingSpeak.begin(client);
}

void loop() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH); // Measure pulse duration
  float distance = (duration * 0.034) / 2; // Calculate distance in cm

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  if (distance > 0 && distance <= 20) {
    digitalWrite(LED_PIN, HIGH);
    Serial.println("Obstacle detected! LED ON.");
    digitalWrite(BUZZER_PIN, HIGH);
  } else {
    digitalWrite(LED_PIN, LOW);
    Serial.println("No obstacle. LED OFF.");
    digitalWrite(BUZZER_PIN, LOW);
  }

  int statusCode = ThingSpeak.writeField(myChannelNo, 1, distance, myWriteAPIkey);
  if (statusCode == 200) {
    Serial.println("Successfully updated ThingSpeak");
  } else {
    Serial.print("Failed to update ThingSpeak. HTTP error code: ");
    Serial.println(statusCode);
  }

  delay(2000);
}


=====================PulseSensor===================
#define PULSE_PIN A0  // Pulse Sensor Pin
#define LED_PIN 13    // LED Pin (Built-in on most Arduino boards)

int threshold = 520;  // Threshold for detecting a heartbeat
int BPM = 0;          // Beats per minute
unsigned long lastBeatTime = 0; // Last detected beat time

void setup() {
  Serial.begin(9600);
  pinMode(PULSE_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
}

void loop() {
  int signal = analogRead(PULSE_PIN);           // Read pulse sensor value
  unsigned long currentTime = millis();        // Get current time

  if (signal > threshold && (currentTime - lastBeatTime) > 250) { // Detect a beat
    lastBeatTime = currentTime;                // Update last beat time
    BPM = 60000 / (currentTime - lastBeatTime); // Calculate BPM
    Serial.print("BPM: ");
    Serial.println(BPM);

    digitalWrite(LED_PIN, BPM > 100 ? HIGH : LOW); // Turn LED ON/OFF based on BPM
  }
}


=====================GasSensor===================
#define GAS_SENSOR_PIN A0 // Analog pin for Gas Sensor
#define LED_PIN 13        // LED pin (built-in on most Arduino boards)

int smokeThreshold = 350;

void setup() {
  Serial.begin(9600);      // Start Serial communication
  pinMode(GAS_SENSOR_PIN, INPUT); // Set Gas Sensor pin as input
  pinMode(LED_PIN, OUTPUT);       // Set LED pin as output
  digitalWrite(LED_PIN, LOW);     // Ensure LED is off initially
}

void loop() {
  int smokeValue = analogRead(GAS_SENSOR_PIN); // Read gas sensor value
  Serial.print("Smoke Value: ");
  Serial.println(smokeValue);                 // Print smoke value to Serial Monitor

  if (smokeValue > smokeThreshold) {
    digitalWrite(LED_PIN, HIGH);              // Turn LED ON if smoke value exceeds threshold
    Serial.println("Warning: High Smoke Level! LED ON");
  } else {
    digitalWrite(LED_PIN, LOW);               // Turn LED OFF if smoke value is below threshold
    Serial.println("Smoke Level Normal. LED OFF");
  }

  delay(500); // Short delay for stable readings
}

=======================PhotoResister=====================
//1st -> power
//2nd -> A0->2nd && GND->RESISTER->2nd

#define LDR_PIN A0   // LDR connected to analog pin A0
#define LED_PIN 13   // LED connected to digital pin 13 (built-in)

int lightThreshold = 500; // Adjust this value based on your environment

void setup() {
  Serial.begin(9600);      // Initialize Serial Monitor
  pinMode(LDR_PIN, INPUT); // Set LDR pin as input
  pinMode(LED_PIN, OUTPUT); // Set LED pin as output
  digitalWrite(LED_PIN, LOW);
}

void loop() {
  int lightValue = analogRead(LDR_PIN); // Read the value from the LDR
  Serial.print("Light Intensity: ");
  Serial.println(lightValue);

  if (lightValue < lightThreshold) {   // If light intensity is below the threshold
    digitalWrite(LED_PIN, HIGH);       // Turn on LED
    Serial.println("Low light detected! LED ON");
  } else {
    digitalWrite(LED_PIN, LOW);        // Turn off LED
    Serial.println("Bright light detected! LED OFF");
  }
  delay(500);
}

=======================IRsensor=====================
#define IR_SENSOR_PIN D5   // IR sensor output pin connected to D5
#define LED_PIN D4         // LED pin connected to D4 (optional)

void setup() {
  Serial.begin(115200);       // Start the Serial Monitor
  pinMode(IR_SENSOR_PIN, INPUT); // Set the IR sensor pin as input
  pinMode(LED_PIN, OUTPUT);      // Set the LED pin as output
  digitalWrite(LED_PIN, LOW);    // Ensure LED is OFF initially
}

void loop() {
  int sensorValue = digitalRead(IR_SENSOR_PIN); // Read IR sensor value

  if (sensorValue == LOW) { // Object detected (Active LOW sensor)
    Serial.println("Obstacle detected!");
    digitalWrite(LED_PIN, HIGH); // Turn ON LED
  } else {
    Serial.println("No obstacle.");
    digitalWrite(LED_PIN, LOW);  // Turn OFF LED
  }

  delay(100); // Small delay for stability
}
