create table customer (
    Custid varchar(20) primary key check(Custid like 'C%'),
    Custname varchar(20),
    Address varchar(50),
    Birthdata date );

create table account(
    Accno int primary key,
    Custid varchar(20) references customer(Custid),
    Acctype varchar(20),
    Acc_balance int );

create table transaction(
    Transid int primary key,
    Accno int references account(Accno),
    Transdate date,
    Transamount int );

create or replace procedure display_accounts_with_balance
IS
begin
    for account_rec IN (select c.Custid,c.Custname,a.Accno,a.Acctype,a.Acc_balance from customer c join account a on c.Custid = a.Custid 
    where a.Acc_balance > 40000)
    loop
        DBMS_OUTPUT.PUT_LINE('Customer ID: ' || account_rec.Custid);
        DBMS_OUTPUT.PUT_LINE('Customer Name: ' || account_rec.Custname);
        DBMS_OUTPUT.PUT_LINE('Account Number: ' || account_rec.Accno);
        DBMS_OUTPUT.PUT_LINE('Account Type: ' || account_rec.Acctype);
        DBMS_OUTPUT.PUT_LINE('Account Balance: ' || account_rec.Acc_balance);
        DBMS_OUTPUT.PUT_LINE('------------------------');
    end loop;

    exception
        when NO_DATA_FOUND then
            DBMS_OUTPUT.PUT_LINE('No accounts found with balance > 40000.');
end display_accounts_with_balance;
/

execute display_accounts_with_balance;

create or replace function calculate_total_balance
return number
IS
    total_balance NUMBER := 0;
begin
    for account_rec IN (select Acc_balance
                         from account
                         where Acctype = 'saving')
    loop
        total_balance := total_balance + account_rec.Acc_balance;
    end loop;

    return total_balance;
end calculate_total_balance;
/

declare
begin
dbms_output.put_line('Total Balance : '||calculate_total_balance());
end;

create or replace trigger prevent_nagpur_customer
before insert on customer
for each row
declare
begin
    if :NEW.city like '%Nagpur%' then
        RAISE_APPLICATION_ERROR(-20001, 'Cannot insert customers from Nagpur.');
    end if;
end prevent_nagpur_customer;
/
